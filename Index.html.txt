        // FORMATTERS & HELPERS

        function formatPKR(amount) {
            const val = Math.abs(amount).toLocaleString('en-PK');
            return `${amount < 0 ? '-' : ''}PKR ${val}`;
        }

        function getCategoryObj(catId) {
            return CATEGORIES.find(c => c.id === catId) || { name: catId, icon: 'tag', color: '#64748b' };
        }

        function getAccountObj(accId) {
            return appState.accounts.find(a => a.id === accId) || { name: 'Unknown', bankName: '' };
        }

        function getCurrentMonthTransactions() {
            const now = new Date();
            const currentYear = now.getFullYear();
            const currentMonth = now.getMonth();

            return appState.transactions.filter(t => {
                const d = new Date(t.date);
                return d.getFullYear() === currentYear && d.getMonth() === currentMonth;
            });
        }

        // EXPENSE DETAILS & REPORTS VIEW RENDERER
        // ==========================================
        function renderExpenseDetailsView() {
            const currentMonthTx = getCurrentMonthTransactions();

            const totalIncome = currentMonthTx
                .filter(t => t.type === 'income')
                .reduce((sum, t) => sum + Number(t.amount), 0);

            const totalExpense = currentMonthTx
                .filter(t => t.type === 'expense')
                .reduce((sum, t) => sum + Number(t.amount), 0);

            // Group Expenses by Category
            const categorySums = {};
            CATEGORIES.forEach(c => categorySums[c.id] = 0);
            
            currentMonthTx.filter(t => t.type === 'expense').forEach(t => {
                categorySums[t.category] = (categorySums[t.category] || 0) + Number(t.amount);
            });

            return `
                <div class="space-y-4">
                    <!-- SECTION 1: INCOME VS EXPENSE SUMMARY CARD -->
                    <div class="bg-white dark:bg-slate-800 p-4 rounded-3xl border border-slate-200 dark:border-slate-800 shadow-sm space-y-3">
                        <h2 class="text-xs font-bold uppercase tracking-wider text-slate-400">Month Income vs Expense Summary</h2>
                        <div class="grid grid-cols-2 gap-3">
                            <div class="bg-emerald-50 dark:bg-emerald-950/40 p-3.5 rounded-2xl border border-emerald-200/50 dark:border-emerald-900/50">
                                <div class="flex items-center space-x-1.5 text-emerald-600 dark:text-emerald-400 mb-1">
                                    <i data-lucide="arrow-down-left" class="w-4 h-4"></i>
                                    <span class="text-xs font-semibold">Total Income</span>
                                </div>
                                <p class="text-base font-extrabold text-emerald-700 dark:text-emerald-300">${formatPKR(totalIncome)}</p>
                            </div>

                            <div class="bg-rose-50 dark:bg-rose-950/40 p-3.5 rounded-2xl border border-rose-200/50 dark:border-rose-900/50">
                                <div class="flex items-center space-x-1.5 text-rose-600 dark:text-rose-400 mb-1">
                                    <i data-lucide="arrow-up-right" class="w-4 h-4"></i>
                                    <span class="text-xs font-semibold">Total Expense</span>
                                </div>
                                <p class="text-base font-extrabold text-rose-700 dark:text-rose-300">${formatPKR(totalExpense)}</p>
                            </div>
                        </div>

                        <!-- Savings Rate Indicator Bar -->
                        <div class="pt-2">
                            <div class="flex justify-between text-xs text-slate-500 mb-1 font-medium">
                                <span>Net Savings Ratio</span>
                                <span>${totalIncome > 0 ? Math.max(0, Math.round(((totalIncome - totalExpense) / totalIncome) * 100)) : 0}%</span>
                            </div>
                            <div class="w-full h-2 rounded-full bg-slate-100 dark:bg-slate-700 overflow-hidden flex">
                                <div class="h-full bg-emerald-500 transition-all duration-500" style="width: ${totalIncome > 0 ? Math.min(100, Math.max(0, ((totalIncome - totalExpense) / totalIncome) * 100)) : 0}%"></div>
                            </div>
                        </div>
                    </div>

                    <!-- SECTION 2: DETAILED CATEGORY EXPENSE LIST -->
                    <div class="bg-white dark:bg-slate-800 p-4 rounded-3xl border border-slate-200 dark:border-slate-800 shadow-sm space-y-3">
                        <h2 class="text-sm font-bold text-slate-800 dark:text-slate-100">Category Wise Breakdown</h2>
                        <div class="space-y-3 pt-1">
                            ${CATEGORIES.map(cat => {
                                const spent = categorySums[cat.id] || 0;
                                const percentage = totalExpense > 0 ? ((spent / totalExpense) * 100).toFixed(1) : 0;
                                return `
                                    <div class="space-y-1.5">
                                        <div class="flex items-center justify-between text-xs">
                                            <div class="flex items-center space-x-2">
                                                <div class="w-7 h-7 rounded-lg flex items-center justify-center text-white" style="background-color: ${cat.color}">

                                                <span class="font-bold text-slate-900 dark:text-white">${formatPKR(spent)}</span>
                                                <span class="text-[10px] text-slate-400 ml-1">(${percentage}%)</span>
                                            </div>
                                        </div>
                                        <div class="w-full h-1.5 rounded-full bg-slate-100 dark:bg-slate-700 overflow-hidden">
                                            <div class="h-full rounded-full transition-all duration-300" style="width: ${percentage}%; background-color: ${cat.color}"></div>
                                        </div>
                                    </div>
                                `;
                            }).join('')}
                        </div>
                    </div>
                </div>
            `;
        }

        function showModal(htmlContent) {
            const backdrop = document.getElementById('modal-backdrop');
            const panel = document.getElementById('modal-panel');
            const content = document.getElementById('modal-content');

            content.innerHTML = htmlContent;
            backdrop.classList.remove('hidden');

        }

        function closeModal() {
            const backdrop = document.getElementById('modal-backdrop');
            const panel = document.getElementById('modal-panel');

            panel.classList.add('translate-y-full');
            backdrop.classList.add('opacity-0');
            setTimeout(() => {
                backdrop.classList.add('hidden');
            }, 300);
        }

        function handleCreateAccount(e) {
            e.preventDefault();
            const type = document.getElementById('acc-type-select').value;
            const bankName = type === 'Bank' ? document.getElementById('acc-bank-name').value : '';
            const name = document.getElementById('acc-name').value;
            let rawBalance = parseFloat(document.getElementById('acc-balance').value) || 0;
            const sign = document.querySelector('input[name="acc-sign"]:checked').value;
            
            const balance = sign === 'negative' ? -Math.abs(rawBalance) : Math.abs(rawBalance);

            const newAccount = {
                id: 'acc_' + Date.now(),
                name,
                type,
                bankName,
                balance,
                active: true,
                isNegative: sign === 'negative'
            };

            appState.accounts.push(newAccount);
            saveState();
            closeModal();
            renderCurrentTab();
            showToast('Account added successfully');
        }

            document.getElementById('tx-type-input').value = type;

            [btnExpense, btnIncome, btnTransfer].forEach(b => b.className = "py-2 rounded-lg font-bold text-xs text-slate-600 dark:text-slate-400 transition");

                singleSec.classList.remove('hidden');
                catSec.classList.remove('hidden');

                btnIncome.className = "py-2 rounded-lg font-bold text-xs bg-white dark:bg-slate-700 text-emerald-600 shadow-sm transition";
                singleSec.classList.remove('hidden');
                catSec.classList.remove('hidden');
                transferSec.classList.add('hidden');
            } else if (type === 'transfer') {
                btnTransfer.className = "py-2 rounded-lg font-bold text-xs bg-white dark:bg-slate-700 text-indigo-600 shadow-sm transition";
                singleSec.classList.add('hidden');
                catSec.classList.add('hidden');
                transferSec.classList.remove('hidden');
            }
        }

        function handleCreateTransaction(e) {
            e.preventDefault();
            const type = document.getElementById('tx-type-input').value;
            const amount = parseFloat(document.getElementById('tx-amount').value);
            const date = document.getElementById('tx-date').value;
            const note = document.getElementById('tx-note').value;

            if (!amount || amount <= 0) {
                showToast('Please enter a valid amount');
                return;
            }

            if (type === 'transfer') {
                const fromId = document.getElementById('tx-account-from').value;
                const toId = document.getElementById('tx-account-to').value;

                if (fromId === toId) {
                    showToast('Source and Destination accounts must be different');
                    return;
                }

                const fromAcc = getAccountObj(fromId);
                const toAcc = getAccountObj(toId);

                fromAcc.balance -= amount;
                toAcc.balance += amount;

                appState.transactions.unshift({
                    id: 'tx_' + Date.now(),
                    type: 'transfer',
                    amount,
                    category: 'other',
                    accountId: fromId,
                    toAccountId: toId,
                    date,
                    note: note || `Transfer to ${toAcc.name}`
                });
            } else {
                const category = document.getElementById('tx-category-select').value;
                const accountId = document.getElementById('tx-account-select').value;
                const acc = getAccountObj(accountId);

                if (type === 'expense') {
                    acc.balance -= amount;
                } else if (type === 'income') {
                    acc.balance += amount;
                }

                appState.transactions.unshift({
                    id: 'tx_' + Date.now(),
                    type,
                    amount,
                    category,
                    accountId,
                    date,
                    note
                });
            }

            saveState();
            closeModal();
            renderCurrentTab();
            showToast('Transaction saved');
        }

        function saveSyncSettings() {
            const url = document.getElementById('sync-url-input').value.trim();
            appState.syncUrl = url;
            saveState();
            if (url) {
                triggerGoogleSheetsSync();
            } else {
                showToast('Sync settings saved');
            }
        }

        function toggleAutoSync(checked) {
            appState.autoSync = checked;
            saveState();
        }

        function triggerGoogleSheetsSync() {
            if (!appState.syncUrl) return;

            const indicator = document.getElementById('sync-indicator');
            if (indicator) indicator.classList.remove('hidden');

            fetch(appState.syncUrl, {
                method: 'POST',
                mode: 'no-cors',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(appState)
            }).then(() => {
                showToast('Synced to Google Sheets');
            }).catch(err => {
                console.error(err);
                showToast('Sync failed');
            }).finally(() => {
                if (indicator) indicator.classList.add('hidden');
            });
        }

        function exportJSONBackup() {
            const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(appState, null, 2));
            const downloadAnchor = document.createElement('a');
            downloadAnchor.setAttribute("href", dataStr);
            downloadAnchor.setAttribute("download", `ExpenseTracker_Backup_${new Date().toISOString().split('T')[0]}.json`);
            document.body.appendChild(downloadAnchor);
            downloadAnchor.click();
            downloadAnchor.remove();
        }

        function importJSONBackup(event) {
            const fileReader = new FileReader();
            fileReader.onload = function(e) {
                try {
                    const parsed = JSON.parse(e.target.result);
                    if (parsed && parsed.accounts && parsed.transactions) {
                        appState = parsed;
                        saveState();
                        closeModal();
                        renderCurrentTab();
                        showToast('Backup restored successfully!');
                    } else {
                        showToast('Invalid backup file structure');
                    }
                } catch (err) {
                    showToast('Error reading backup file');
                }
            };
            fileReader.readAsText(event.target.files[0]);
        }

        function exportCSV() {
            let csvContent = "data:text/csv;charset=utf-8,ID,Type,Amount (PKR),Category,Account,Date,Note\n";
            appState.transactions.forEach(t => {
                const acc = getAccountObj(t.accountId);
                const cat = getCategoryObj(t.category);
                csvContent += `"${t.id}","${t.type}","${t.amount}","${cat.name}","${acc.name}","${t.date}","${t.note || ''}"\n`;
            });

            const encodedUri = encodeURI(csvContent);
            const link = document.createElement("a");
            link.setAttribute("href", encodedUri);
            link.setAttribute("download", `Transactions_${new Date().toISOString().split('T')[0]}.csv`);
            document.body.appendChild(link);
            link.click();
            link.remove();
        }
